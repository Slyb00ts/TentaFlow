# test_checkout.py — E2E: płatność kartą w koszyku B2B import re import pytest from playwright.sync_api import Page, expect @pytest.fixture def logged_in(page: Page, base_url, test_password): page.goto(f"{base_url}/logowanie") page.get_by_label("E-mail").fill("anna.kowalska@euvic.pl") page.get_by_label("Hasło").fill(test_password) page.get_by_role("button", name="Zaloguj").click() expect(page.get_by_test_id("user-menu")).to_be_visible() return page @pytest.mark.tags("checkout", "płatności") def test_platnosc_karta_visa(logged_in: Page, base_url): page = logged_in page.goto(f"{base_url}/koszyk") page.get_by_role("button", name="Do kasy").click() expect(page).to_have_url(re.compile(r".*/checkout")) # Dane karty testowej (Visa) page.get_by_label("Numer karty").fill("4111 1111 1111 1111") page.get_by_label("Data ważności").fill("12/28") page.get_by_label("CVC").fill("123") page.get_by_role("button", name="Zapłać").click() # Oczekiwany rezultat: zamówienie opłacone expect(page.get_by_test_id("order-status")).to_have_text("Opłacone") order_id = page.get_by_test_id("order-id").inner_text() assert order_id.startswith("ZAM-")
