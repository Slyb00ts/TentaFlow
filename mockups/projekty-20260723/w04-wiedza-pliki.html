"https://api.portal-b2b.pl/v2"; const MAX_RETRIES = 3; export interface RequestOptions { method?: "GET" | "POST" | "PUT" | "DELETE"; body?: unknown; retries?: number; } export async function request<T>(path: string, opts: RequestOptions = {}): Promise<T> { const token = await getAuthToken(); const res = await fetch(`${BASE_URL}${path}`, { method: opts.method ?? "GET", headers: { "Content-Type": "application/json", Authorization: `Bearer ${token}`, }, body: opts.body ? JSON.stringify(opts.body) : undefined, }); if (res.status === 401 && (opts.retries ?? 0) < MAX_RETRIES) { await refreshToken(); return request<T>(path, { ...opts, retries: (opts.retries ?? 0) + 1 }); } if (!res.ok) throw new ApiError(res.status, await res.text()); return res.json() as Promise<T>; }
Jak to trafia do wiedzy?
Pliki repozytorium są dzielone na chunki i zapisywane w bazie wiedzy projektu. Agenci i chat korzystają z nich jako kontekstu. Wybierz plik w drzewie, aby go podejrzeć. Akcją „Wyklucz z indeksu" pominiesz plik przy kolejnej indeksacji (np. wygenerowany kod), a „Indeksuj ponownie" przetworzy go jeszcze raz. Zmiany w repozytorium pobierzesz przyciskiem Odśwież (fetch + delta) — reindeksowane są tylko zmienione pliki.
